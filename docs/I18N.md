> **Layihə:** Cahan Academy Dashboard  
> **GitHub:** [ismetcahangirov/Cahan-Academy-Dashboard](https://github.com/ismetcahangirov/Cahan-Academy-Dashboard)

---

## 1. Ümumi Baxış

| Parametr | Dəyər |
|---|---|
| Kitabxana | `i18next` + `react-i18next` |
| Default dil | Azərbaycan (`az`) |
| Dəstəklənən dillər | AZ, EN, RU |
| Saxlama yeri | `localStorage` (`i18nextLng` açarı) |
| Namespace | `translation` (tək namespace) |
| Tarix/Saat formatı | `date-fns` ilə lokalizasiya |

---

## 2. Quraşdırma

```bash
npm install i18next react-i18next i18next-browser-languagedetector
npm install date-fns
```

---

## 3. Konfiqurasiya

```javascript
// src/i18n/index.js

import i18n                         from 'i18next';
import { initReactI18next }         from 'react-i18next';
import LanguageDetector             from 'i18next-browser-languagedetector';

import az from './az.json';
import en from './en.json';
import ru from './ru.json';

i18n
  .use(LanguageDetector)   // Brauzerin dilini avtomatik tanı
  .use(initReactI18next)
  .init({
    resources: {
      az: { translation: az },
      en: { translation: en },
      ru: { translation: ru },
    },
    fallbackLng:   'az',        // Tapılmasa AZ-a dön
    defaultNS:     'translation',
    interpolation: {
      escapeValue: false,       // React özü XSS-dən qoruyur
    },
    detection: {
      order:  ['localStorage', 'navigator'],
      caches: ['localStorage'],
      lookupLocalStorage: 'i18nextLng',
    },
  });

export default i18n;
```

```jsx
// src/main.jsx
import './i18n/index.js'; // i18n-i tətbiqin ən əvvəlində import et

import React        from 'react';
import ReactDOM     from 'react-dom/client';
import { Provider } from 'react-redux';
import { store }    from './app/store';
import App          from './App';

ReactDOM.createRoot(document.getElementById('root')).render(
  <React.StrictMode>
    <Provider store={store}>
      <App />
    </Provider>
  </React.StrictMode>
);
```

## 5. Komponentdə İstifadə

### `useTranslation` hook-u

```jsx
import { useTranslation } from 'react-i18next';

const TeachersPage = () => {
  const { t } = useTranslation();

  return (
    <div>
      <h1>{t('teachers.title')}</h1>
      <p>{t('teachers.subtitle', { count: 12 })}</p>
      <button>{t('teachers.invite')}</button>
    </div>
  );
};
```

```jsx
// az.json: "welcome": "Xoş gəlmisiniz, {{name}}!"

// → "Xoş gəlmisiniz, Əli!"
```

## 6. LanguageSwitcher Komponenti

```jsx
// src/components/ui/LanguageSwitcher.jsx

import { useTranslation } from 'react-i18next';
import { Globe }          from 'lucide-react';
import { useState }       from 'react';

const LANGUAGES = [
  { code: 'az', label: 'Azərbaycan', short: 'AZ' },
  { code: 'en', label: 'English',    short: 'EN' },
  { code: 'ru', label: 'Русский',    short: 'RU' },
];

const LanguageSwitcher = () => {
  const { i18n } = useTranslation();
  const [open, setOpen]   = useState(false);
  const current = LANGUAGES.find((l) => l.code === i18n.language) || LANGUAGES[0];

  const handleChange = (code) => {
    i18n.changeLanguage(code);
    setOpen(false);
  };

  return (
    <div className="relative">
      <button
        onClick={() => setOpen((p) => !p)}
        className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg
                   text-sm font-medium text-gray-700 hover:bg-gray-100 transition"
      >
        <Globe size={16} />
        {current.short}
      </button>

      {open && (
        <div className="absolute right-0 top-full mt-1 w-40 bg-white
                        rounded-lg shadow-dropdown border border-gray-100 z-50">
          {LANGUAGES.map((lang) => (
            <button
              key={lang.code}
              onClick={() => handleChange(lang.code)}
              className={`w-full text-left px-4 py-2.5 text-sm transition
                ${lang.code === i18n.language
                  ? 'text-primary font-semibold bg-primary-50'
                  : 'text-gray-700 hover:bg-gray-50'
                }`}
            >
              {lang.label}
            </button>
          ))}
        </div>
      )}
    </div>
  );
};

export default LanguageSwitcher;
```

## 7. Tarix Lokalizasiyası

```javascript
// utils/formatDate.js

import { format, formatDistanceToNow } from 'date-fns';
import { az as azLocale, enUS, ru }    from 'date-fns/locale';
import i18n                             from '../i18n';

const LOCALES = { az: azLocale, en: enUS, ru };

const getLocale = () => LOCALES[i18n.language] || azLocale;

// Tarix formatı: "15 yanvar 2026"
export const formatDate = (date) =>
  format(new Date(date), 'd MMMM yyyy', { locale: getLocale() });

// Tarix + saat: "15 yanvar 2026, 09:00"
export const formatDateTime = (date) =>
  format(new Date(date), 'd MMMM yyyy, HH:mm', { locale: getLocale() });

// Nisbi vaxt: "3 gün əvvəl"
export const timeAgo = (date) =>
  formatDistanceToNow(new Date(date), { addSuffix: true, locale: getLocale() });

// Yalnız saat: "09:00"
export const formatTime = (time) => time; // "09:00" string kimi saxlanır
```

```jsx
// İstifadə nümunəsi
import { formatDate, formatDateTime, timeAgo } from '../utils/formatDate';

<span>{formatDate(homework.dueDate)}</span>
// AZ: "10 fevral 2026"
// EN: "February 10, 2026"
// RU: "10 февраля 2026"

<span>{timeAgo(invitation.createdAt)}</span>
// AZ: "3 gün əvvəl"
// EN: "3 days ago"
// RU: "3 дня назад"
```
